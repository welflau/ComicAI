## 1. 系统概述

### 1.1 核心概念

**无限画布** 是一个可无限延展的二维工作空间，用户可以在上面自由拖拽、排列、连接各种**节点**来构建完整的工作流程。

**节点工作流** 是通过可视化的节点图来表示的数据处理流水线，每个节点执行特定的功能，通过连线传递数据。

## 3. 节点系统设计

### 3.1 节点基础结构

```typescript
// 节点定义
interface Node {
  id: string;                    // 唯一标识
  type: NodeType;                // 节点类型
  position: { x: number; y: number };  // 画布位置
  size: { width: number; height: number };  // 节点尺寸
  data: NodeData;                // 节点数据
  inputs: Port[];               // 输入端口
  outputs: Port[];              // 输出端口
  style?: NodeStyle;            // 样式覆盖
  locked?: boolean;            // 是否锁定
  visible?: boolean;          // 是否可见
}

// 节点类型
type NodeType = 
  | 'text'          // 文本输入
  | 'image'        // 图像输入
  | 'video'       // 视频输入
  | 'audio'       // 音频输入
  | 'script'      // 剧本
  | 'shot'       // 分镜
  | 'generate-image' // 图像生成
  | 'generate-video' // 视频生成
  | 'generate-audio' // 音频生成
  | ' upscale'    // 超分放大
  | 'interpolate' // 补帧
  | 'edit'       // 编辑
  | 'crop'       // 裁剪
  | 'transition' // 转场
  | 'composite'  // 合成
  | 'export'    // 导出
  | 'agent'      // Agent 执行
  | 'group';    // 分组

// 端口定义
interface Port {
  id: string;
  name: string;
  type: 'text' | 'image' | 'video' | 'audio' | 'any';
  direction: 'input' | 'output';
  required?: boolean;
  defaultValue?: any;
}

// 节点数据
interface NodeData {
  label?: string;              // 显示标签
  model?: string;              // 模型选择
  prompt?: string;           // 提示词
  negativePrompt?: string;       // 负向提示词
  parameters?: Record<string, any>;  // 参数
  [key: string]: any;
}
```

### 5.2 拓扑排序算法

```typescript
// 拓扑排序示例
function topologicalSort(nodes: Node[], edges: Edge[]): Node[][] {
  // 1. 构建依赖图
  const graph = buildGraph(nodes, edges);
  
  // 2. 计算入度
  const inDegree = calculateInDegree(nodes, edges);
  
  // 3. 层级分配
  const levels: Node[][] = [];
  
  while (hasUnprocessedNodes(nodes)) {
    const currentLevel = [];
    
    for (const node of nodes) {
      if (inDegree[node.id] === 0 && !node.processed) {
        currentLevel.push(node);
      }
    }
    
    // 移除当前层节点，更新入度
    for (const node of currentLevel) {
      node.processed = true;
      for (const next of graph[node.id]) {
        inDegree[next.id]--;
      }
    }
    
    levels.push(currentLevel);
  }
  
  return levels;
}
```

### 6.2 Undo/Redo 实现

```typescript
// 历史记录管理
class HistoryManager {
  private past: State[] = [];
  private future: State[] = [];
  private maxSize = 50;
  
  // 保存快照
  push(state: State) {
    this.past.push(state);
    if (this.past.length > this.maxSize) {
      this.past.shift();
    }
    this.future = [];  // 清除未来状态
  }
  
  // 撤销
  undo(): State | null {
    const state = this.past.pop();
    if (state) {
      this.future.push(this.currentState);
      return state;
    }
    return null;
  }
  
  // 重做
  redo(): State | null {
    const state = this.future.pop();
    if (state) {
      this.past.push(this.currentState);
      return state;
    }
    return null;
  }
}
```

### 8.2 React Flow 配置

```typescript
import ReactFlow, {
  Background,
  Controls,
  MiniMap,
  useNodesState,
  useEdgesState,
  addEdge,
} from 'reactflow';
import 'reactflow/dist/style.css';

// 节点类型注册
const nodeTypes = {
  text: TextNode,
  image: ImageNode,
  generateImage: GenerateImageNode,
  generateVideo: GenerateVideoNode,
  export: ExportNode,
};

// 画布组件
function Canvas() {
  const [nodes, setNodes, onNodesChange] = useNodesState(initialNodes);
  const [edges, setEdges, onEdgesChange] = useEdgesState(initialEdges);
  
  return (
    <ReactFlow
      nodes={nodes}
      edges={edges}
      onNodesChange={onNodesChange}
      onEdgesChange={onEdgesChange}
      onConnect={(params) => setEdges((eds) => addEdge(params, eds))}
      nodeTypes={nodeTypes}
      fitView
    >
      <Background />
      <Controls />
      <MiniMap />
    </ReactFlow>
  );
}
```
